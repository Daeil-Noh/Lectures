pandas로: 히스토그램 · 상자그림 · IQR 규칙

s = df['strength']
s.hist(bins=30)                      # 히스토그램
df.boxplot(column='strength', by='shift')  # 주야간 비교

q1 = s.quantile(0.25)                # Q1
q3 = s.quantile(0.75)                # Q3
iqr = q3 - q1
low = q1 - 1.5 * iqr                 # 아래 울타리
high = q3 + 1.5 * iqr                # 위 울타리
out = df[(s < low) | (s > high)]     # 울타리 밖 행
print(round(low, 2), round(high, 2), len(out))

def prepare(X):                    # 밝기 보정: 자기 평균 대비 밝기
    return X / X.mean(axis=1, keepdims=True) - 1

X_train, X_test, y_train, y_test = train_test_split(
    prepare(X), y, test_size=0.25, random_state=0, stratify=y)

model = MLPClassifier(hidden_layer_sizes=(128,), max_iter=500, random_state=0)
model.fit(X_train, y_train)
print('시험 정확도:', round(model.score(X_test, y_test), 3))
print(confusion_matrix(y_test, model.predict(X_test)))

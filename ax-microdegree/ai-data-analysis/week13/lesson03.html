def generate(start='<시작>', temperature=1.0, max_words=10):
    word, out = start, ([] if start == '<시작>' else [start])
    for _ in range(max_words):
        p = next_word_probs(word)                    # 다음 단어 확률
        if not p:
            break
        w = np.array(list(p.values())) ** (1 / temperature)   # 온도 적용
        word = rng.choice(list(p.keys()), p=w / w.sum())    # 확률대로 뽑기
        out.append(word)                             # 이어 붙이기
        if word == '.':
            break
    return ' '.join(out)

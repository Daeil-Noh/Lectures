import pandas as pd
DATA = 'https://raw.githubusercontent.com/…/data/'  # 전체 주소는 실습 파일에
df = pd.read_csv(DATA + 'process.csv')

features = ['temp', 'pressure', 'speed', 'humidity', 'thickness']
X = df[features]        # 입력: 공정 조건 5개
y = df['defect']        # 정답: 0 정상, 1 불량
print(y.value_counts())

① 데이터 준비

stratify=y: 정답 비율을 지키며 나누기

from sklearn.model_selection import train_test_split
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.3, random_state=42,
    stratify=y)        # y의 0/1 비율대로 나눈다
print(len(X_train), len(X_test))
print(round(y_train.mean(), 3), round(y_test.mean(), 3))
345 149
0.136 0.141
훈련 345개 (불량 47개)
나무가 질문을 고르는 데 쓴다
시험 149개 (불량 21개)
끝까지 숨겨 두었다가 실력 평가에만
불량률 훈련 13.6% · 시험 14.1% — 전체(13.8%)와 거의 같다. 9~10주는 8:2였지만 오늘은 7:3: 20%면 시험 불량이 14개뿐이라 불량을 잡는 실력을 재기에 너무 적다.
AI와 데이터 분석 · 11주 2차시 · 실습 파일 4단계 · y.mean() = 1의 비율 = 불량률

② 학습과 나무 읽기

DecisionTreeClassifier: 만들고, 학습하고, 점수 보기

from sklearn.tree import DecisionTreeClassifier
model = DecisionTreeClassifier(max_depth=3,
                               random_state=42)
model.fit(X_train, y_train)            # 학습
print(round(model.score(X_train, y_train), 3))  # 훈련
print(round(model.score(X_test, y_test), 3))    # 시험

new = pd.DataFrame([
    [210, 80, 1200, 55, 2.00],   # A: 평범한 조건
    [226, 80, 1200, 60, 1.98],   # B: 고온, 얇음
    [212, 82, 1150, 79, 2.01],   # C: 고습
], columns=features)
print(model.predict(new))
print(model.predict_proba(new).round(3))

predict → 0 또는 1
A 정상 · B 불량 · C 불량
predict_proba → [정상 확률, 불량 확률]
B는 불량 확률 0.857 = 도착한 잎의 훈련 데이터 제품 7개 중 불량 6개
predict는 불량 확률이 0.5를 넘으면 1로 판정합니다. 이 0.5를 바꾸는 이야기가 다음 차시의 판정 기준.
